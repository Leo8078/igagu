/*
 * 정사각형의 길일ㄹ 입력하면 출력하는 문제
 * 예)
 * n = 5일 경우
 * =====
 * =====
 * =====
 * =====
 * =====  
 */

void main() {
  var n = 5;
  var result = "";
  for (var y = 0; y < n; y++) {
    for (var x = 0; x < n; x++) {
      result += "=";
    }
    result += "\n";
  }
  print(result);
}
