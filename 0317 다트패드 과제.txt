/*

꽉 찬 사각형

void main() {
  var n = 10;
  var result = "";
  for (var y = 0; y < n; y++) {
    for (var x = 0; x < n; x++) {
      result += "=";
    }
    result += "\n";
  }
  print(result);
}


테두리 사각형

void main() {
  var n = 10;
  var result = "";
  for (var y = 0; y < n; y++) {
    for (var x = 0; x < n; x++) {
      var c = 
        y == 0 || y == n - 1 || 
        x == 0 || x == n - 1;
      if (c) {
        result += "=";
      } else {
        result += " ";
      }
    }
    result += "\n";
  }
  print(result);
}



X표시 사각형

void main() {
  var n = 10;
  var result = "";
  for (var y = 0; y < n; y++) {
    for (var x = 0; x < n; x++) {
      var c = 
        y == 0 || y == n - 1 || 
        x == 0 || x == n - 1 || 
        x == y || x == n - y - 1; // 추가된 X 표시 조건
      
      if (c) {
        result += "=";
      } else {
        result += " ";
      }
    }
    result += "\n";
  }
  print(result);
}

void main() {
  var input = '2025-03-15'; // YYYY-MM-DD 형식의 입력값

  var parts = input.split('-');
  int year = int.parse(parts[0]);
  int month = int.parse(parts[1]);
  int day = int.parse(parts[2]);

  DateTime date = DateTime(year, month, day);
  List<String> weekdays = ["월요일", "화요일", "수요일", "목요일", "금요일", "토요일", "일요일"];

  print("입력한 날짜는 ${weekdays[date.weekday - 1]}입니다.");
}
